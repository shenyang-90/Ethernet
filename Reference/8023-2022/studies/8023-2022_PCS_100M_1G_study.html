1. 100BASE-X PCS (Clause 24) — 4B/5B Encoding

1.1 Overview

100BASE-X (100BASE-TX and 100BASE-FX) uses a 4B/5B block coding scheme derived from FDDI (ISO/IEC 9314-1). The PCS maps four-bit MII nibbles into five-bit code-groups, providing:

  • 16 Data code-groups (0x0–0xF)
  • Control code-groups: Idle (/I/), SSD (/J//K/), ESD (/T//R/), SLEEP (/P/), Error (/H/), Invalid (/V/)
  • 80% coding efficiency: 100 Mb/s data rate on a 125 Mb/s physical channel
  • Sufficient transition density for clock recovery

1.5 PCS Transmit State Machine (Pseudocode)

// Constants
SSD1 = 5'b11000;   // /J/
SSD2 = 5'b10001;   // /K/
ESD1 = 5'b01101;   // /T/
ESD2 = 5'b00111;   // /R/
IDLE = 5'b11111;   // /I/
HALT = 5'b00100;   // /H/
SLEEP = 5'b00000;  // /P/ (EEE)

// States: BEGIN, IDLE, START_STREAM_J, START_STREAM_K, TRANSMIT_DATA,
//         TRANSMIT_ERROR, END_STREAM_T, END_STREAM_R, START_ERROR_J,
//         START_ERROR_K, ERROR_CHECK, TX_SLEEP, TX_QUIET, BackToIDLE

BEGIN:
  transmitting <= FALSE;
  COL <= FALSE;
  tx_bits <= IDLE;
  if (TX_EN == TRUE && TX_ER == FALSE) goto START_STREAM_J;
  else if (EEE && TX_EN == FALSE && TX_ER == TRUE && TXD == 4'b0001)
    goto TX_SLEEP;  // LPI entry
  else goto IDLE;

IDLE:
  transmitting <= FALSE;
  COL <= FALSE;
  tx_bits <= IDLE;
  if (TX_EN == TRUE && TX_ER == FALSE) goto START_STREAM_J;
  else if (EEE && TX_EN == FALSE && TX_ER == TRUE && TXD == 4'b0001)
    goto TX_SLEEP;

START_STREAM_J:
  transmitting <= TRUE;
  COL <= receiving;   // collision if receiving while transmitting
  tx_bits <= SSD1;
  if (sentCodeGroup.indicate) goto START_STREAM_K;

START_STREAM_K:
  COL <= receiving;
  tx_bits <= SSD2;
  if (sentCodeGroup.indicate) goto TRANSMIT_DATA;

TRANSMIT_DATA:
  COL <= receiving;
  tx_bits <= ENCODE(TXD<3:0>);
  if (sentCodeGroup.indicate && TX_EN == FALSE && TX_ER == FALSE)
    goto END_STREAM_T;
  else if (sentCodeGroup.indicate && TX_EN == TRUE && TX_ER == TRUE)
    goto TRANSMIT_ERROR;

TRANSMIT_ERROR:
  COL <= receiving;
  tx_bits <= HALT;
  if (sentCodeGroup.indicate && TX_EN == FALSE) goto END_STREAM_T;

END_STREAM_T:
  transmitting <= FALSE;
  COL <= FALSE;
  tx_bits <= ESD1;
  if (sentCodeGroup.indicate) goto END_STREAM_R;

END_STREAM_R:
  tx_bits <= ESD2;
  if (sentCodeGroup.indicate) goto IDLE;

// EEE LPI Transmit States
TX_SLEEP:
  tx_bits <= SLEEP;
  Start lpi_tx_ts_timer;
  if (sentCodeGroup.indicate && lpi_tx_ts_timer_done)
    goto TX_QUIET;
  else if (sentCodeGroup.indicate && TX_EN == TRUE)
    goto START_STREAM_J;  // abort LPI

TX_QUIET:
  tx_quiet <= TRUE;
  Start lpi_tx_tq_timer;
  if (lpi_tx_tq_timer_done) goto TX_SLEEP;  // refresh
  else if (TX_EN == TRUE || TX_ER == FALSE || TXD != TX_LP_IDLE)
    goto BackToIDLE;

BackToIDLE:
  tx_quiet <= FALSE;
  goto IDLE;

1.6 PCS Receive State Machine (Pseudocode)

// States: BEGIN, IDLE, CARRIER_DETECT, IDENTIFY_JK, START_OF_STREAM_J,
//         START_OF_STREAM_K, DATA, END_OF_STREAM, PREMATURE_END,
//         DATA_ERROR, LINK_FAILED, BAD_SSD, RX_SLEEP, RX_QUIET,
//         RX_WAKE, START_RX_QUIET, START_RX_SLEEP, WAIT_IDLE,
//         RX_LPI_LINK_FAIL, WAIT_SLEEP

BEGIN:
  rx_bits[9:0] <= 10'b11111_11111;
  receiving <= FALSE;
  RX_DV <= FALSE;
  RX_ER <= FALSE;
  goto IDLE;

IDLE:
  receiving <= FALSE;
  RX_DV <= FALSE;
  RX_ER <= FALSE;
  if (link_status != OK) goto LINK_FAILED;
  if (rx_bits[0] == 0 && rx_bits[9:2] != 8'b11111111) goto CARRIER_DETECT;

CARRIER_DETECT:
  receiving <= TRUE;
  // Two non-contiguous zeros within 10 bits = carrier
  if (rx_bits[9:0] == {SSD1, SSD2}) goto START_OF_STREAM_J;
  else if (rx_bits[9:0] == {SLEEP, SSD1}) goto IDENTIFY_JK;  // EEE
  else goto DATA_ERROR;  // False carrier

IDENTIFY_JK:  // EEE only
  if (rx_bits[9:0] == {SLEEP, SLEEP}) goto WAIT_SLEEP;
  else goto IDLE;

START_OF_STREAM_J:
  RX_DV <= TRUE;
  RXD <= 4'b0101;   // preamble nibble replacement
  goto START_OF_STREAM_K;

START_OF_STREAM_K:
  RXD <= 4'b0101;   // preamble nibble replacement
  if (gotCodeGroup.indicate) goto DATA;

DATA:
  RX_ER <= FALSE;
  RXD <= DECODE(rx_bits[9:5]);
  if (gotCodeGroup.indicate && rx_bits[9:5] not in DATA) goto DATA_ERROR;
  if (gotCodeGroup.indicate && rx_bits[9:0] == ESD) goto END_OF_STREAM;
  if (gotCodeGroup.indicate && rx_bits[9:0] == IDLES) goto PREMATURE_END;

END_OF_STREAM:
  rx_bits <= 10'b11111_11111;
  receiving <= FALSE;
  goto IDLE;

PREMATURE_END:
  RX_ER <= TRUE;
  receiving <= FALSE;
  goto IDLE;

DATA_ERROR:
  RX_ER <= TRUE;
  if (gotCodeGroup.indicate) goto END_OF_STREAM;

LINK_FAILED:
  RX_ER <= TRUE;
  receiving <= FALSE;
  goto IDLE;

BAD_SSD:
  RX_ER <= TRUE;
  RXD <= 4'b1110;   // False Carrier Indication
  if (rx_bits[9:0] == IDLES) goto IDLE;

// EEE LPI Receive States
WAIT_SLEEP:
  rx_lpi <= TRUE;
  Start lpi_rx_ts_timer;
  if (lpi_rx_ts_timer_done) goto START_RX_QUIET;

START_RX_QUIET:
  rx_quiet <= TRUE;
  Start lpi_rx_tq_timer;
  if (signal_status == OFF || lpi_rx_tq_timer_done)
    goto RX_LPI_LINK_FAIL;
  if (rx_bits == IDLES) goto WAIT_IDLE;

WAIT_IDLE:
  Start lpi_rx_ti_timer;
  if (lpi_rx_ti_timer_done) goto IDLE;
  if (rx_bits != IDLES) goto START_RX_SLEEP;

START_RX_SLEEP:
  Start lpi_rx_ts_timer;
  goto RX_SLEEP;

RX_SLEEP:
  rx_quiet <= FALSE;
  if (rx_bits == IDLES) goto WAIT_IDLE;
  if (signal_status == OFF) goto RX_LPI_LINK_FAIL;

RX_WAKE:
  rx_quiet <= FALSE;
  Start lpi_link_fail_timer;
  if (rx_bits == IDLES) goto IDLE;
  if (lpi_link_fail_timer_done) goto RX_LPI_LINK_FAIL;

RX_LPI_LINK_FAIL:
  lpi_link_fail <= TRUE;
  rx_quiet <= FALSE;
  goto IDLE;

1.7 Carrier Sense and Collision Detection

The Carrier Sense process asserts CRS on the MII when either transmitting or receiving is TRUE:

CARRIER_SENSE_OFF:  CRS <= FALSE;
  if (transmitting == TRUE || receiving == TRUE) goto CARRIER_SENSE_ON;

CARRIER_SENSE_ON:   CRS <= TRUE;
  if (transmitting == FALSE && receiving == FALSE) goto CARRIER_SENSE_OFF;

Collision detection is performed by the Transmit process: COL <= transmitting && receiving. This follows the 10BASE-T model where a reception occurring simultaneously with transmission indicates a collision.

1.8 Code-Group Alignment

The Receive Bits process collects serial code-bits from the PMA into a 10-bit sliding window rx_bits[9:0]. Before alignment:

  • New code-bits shift into rx_bits[0]
  • The process signals gotCodeGroup.indicate every 5 bits once alignment is achieved
  • Alignment is established by detecting the SSD (/J/K/ = 11000 10001)
  • After alignment, code-groups are decoded from rx_bits[9:5] (current) and rx_bits[4:0] (previous)

RTL Implementation Note: 5-Bit Alignment Logic

always_ff @(posedge clk_125mhz) begin
  if (!aligned) begin
    rx_bits <= {rx_bits[8:0], rx_code_bit};
    // Scan for SSD pattern across all 5 possible alignments
    if (rx_bits[9:5] == SSD1 && rx_bits[4:0] == SSD2)
      aligned <= 1'b1;
  end else begin
    // Aligned: process 5 bits at a time
    bit_counter <= bit_counter + 1'b1;
    rx_bits <= {rx_bits[8:0], rx_code_bit};
    if (bit_counter == 3'd4) begin
      gotCodeGroup <= 1'b1;
      bit_counter <= 3'd0;
    end
  end
end

1.9 Code Violation Detection

Code violations are detected when:

  • An invalid code-group is received (any /V/ code)
  • An ESD is not properly formed (/T/ not followed by /R/)
  • An SSD is malformed
  • Two Idle code-groups (/I/I/) appear before an ESD (premature termination)
  • The /H/ code-group appears in the data stream (transmit error propagation)

Invalid code-groups result in RX_ER assertion and may cause the Receive state machine to enter the DATA_ERROR state.

2. 1000BASE-X PCS (Clause 36) — 8B/10B Encoding

2.1 Overview

1000BASE-X is based on ANSI INCITS 230-1994 (Fibre Channel Physical and Signaling Interface). It uses 8B/10B block coding, providing:

  • 256 Data code-groups (/Dx.y/) and 12 Special code-groups (/Kx.y/)
  • DC balance via Running Disparity (RD)
  • High transition density (3–8 transitions per symbol)
  • Comma-based alignment via /K28.5/
  • Ordered sets for frame delineation and link management

2.2 8B/10B Notation

An unencoded octet is labeled ABCD EFGH (bit A = LSB). The 10-bit encoded code-group is labeled abcdei fghj:

  • Data code-group: /Dx.y/ where x = decimal value of EDCBA, y = decimal value of HGF
  • Special code-group: /Kx.y/ where x = decimal value of EDCBA, y = decimal value of HGF
  • Bit transmission order: a, b, c, d, e, i, f, g, h, j (bit a first)

2.3 Running Disparity (RD) Rules

Running Disparity is a binary parameter with values negative (–) or positive (+). Initial RD after reset is negative.

RuleDescription
Sub-block divisionFirst 6 bits (abcdei) = six-bit sub-block; last 4 bits (fghj) = four-bit sub-block
RD+ end conditionSub-block has more ones than zeros; OR six-bit = 000111; OR four-bit = 0011
RD– end conditionSub-block has more zeros than ones; OR six-bit = 111000; OR four-bit = 1100
Neutral sub-blockEqual ones and zeros: RD passes through unchanged
Code-group RDRD at start of six-bit sub-block = RD at end of last code-group; RD at start of four-bit = RD at end of six-bit; RD at end of code-group = RD at end of four-bit
Key Constraint: Sub-blocks 000111 and 0011 are generated only when RD at sub-block start is positive. Sub-blocks 111000 and 1100 are generated only when RD at sub-block start is negative. This ensures proper DC balance and run-length limiting.

2.6 Comma Detection and Synchronization

The comma pattern is a 7-bit string used for code-group alignment:

  • Comma+: 0011111 (appears in /K28.1/, /K28.5/, /K28.7/ when RD–)
  • Comma–: 1100000 (appears in /K28.1/, /K28.5/, /K28.7/ when RD+)

The comma is singular: in the absence of errors, it cannot appear in any other location within a code-group or across code-group boundaries (except the diagnostic /K28.7/ case).

RTL Implementation: Comma Detector

3. 1000BASE-T PCS (Clause 40) — 4D-PAM5

3.1 Overview

1000BASE-T achieves 1 Gb/s over four pairs of Category 5 cabling by transmitting 250 Mb/s per pair using 4D-PAM5 modulation at 125 MBd. Each four-dimensional symbol is a 4-tuple (An, Bn, Cn, Dn) of quinary symbols from the set {–2, –1, 0, +1, +2}.

  • Data rate: 1000 Mb/s aggregate (250 Mb/s per pair)
  • Symbol rate: 125 MBd per pair (8 ns symbol period)
  • Modulation: 5-level PAM (4D-PAM5)
  • Encoding: 8 data bits + 1 convolutional bit → 9 bits → 4 PAM5 symbols
  • Full duplex: Simultaneous TX/RX on same pairs via hybrids and echo cancellers

3.2 Side-Stream Scrambler Polynomials

The PCS Transmit employs side-stream scrambling to ensure uncorrelated symbols and no DC bias. Two different polynomials are used for MASTER and SLAVE:

RolePolynomialLFSR Taps
MASTERgM(x) = 1 + x13 + x33Scrn[0] = Scrn[13] ⊕ Scrn[33]
SLAVEgS(x) = 1 + x20 + x33Scrn[0] = Scrn[20] ⊕ Scrn[33]

The 33-bit shift register Scrn[32:0] generates a maximum-length sequence of period 233–1. The state shall never be initialized to all zeros.

RTL Implementation: 33-bit Side-Stream Scrambler

module scrambler_33bit (
  input  wire        clk,
  input  wire        rst_n,
  input  wire        master_mode,  // 1=MASTER, 0=SLAVE
  output wire        scr_out       // Scrn[0]
);
  logic [32:0] lfsr;
  wire tap = master_mode ? lfsr[13] : lfsr[20];
  
  always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      lfsr <= 33'h1;  // Non-zero seed (implementer-defined)
    end else begin
      lfsr <= {tap ^ lfsr[32], lfsr[32:1]};
    end
  end
  
  assign scr_out = lfsr[0];
endmodule

3.4 Convolutional Encoder

A 4-state convolutional encoder adds 1 bit of redundancy to 8 scrambled data bits, producing 9 bits that are mapped to four PAM5 symbols. The encoder uses a 3-bit state csn[2:0]:

csn[0] = csn-1[2]   // state feedback
csresetn = tx_enablen-2 & ~tx_enablen   // reset at end of frame

// State update (during data transmission)
csn[1] = Sdn[6] ⊕ csn-1[0]   if tx_enablen-2 == 1
       = 0                    else
csn[2] = Sdn[7] ⊕ csn-1[1]   if tx_enablen-2 == 1
       = 0                    else

// Output bit
Sdn[8] = csn[0]

3.10 Auto-Negotiation and Training

1000BASE-T uses Clause 28 Auto-Negotiation with Fast Link Pulses (FLP) and Normal Link Pulses (NLP):

  • FLP: 33 pulse positions (17 clock, 16 data) at 125 µs intervals — advertises capabilities
  • NLP: 10BASE-T compatibility pulse — used for 10/100/1000 negotiation
  • MASTER/SLAVE: Determined during Auto-Negotiation; MASTER uses local clock, SLAVE uses recovered clock (loop timing)

After Auto-Negotiation, the PHY enters training mode:

  1. PMA PHY Control sends SEND_I (idle training symbols)
  2. Receiver trains echo canceller, equalizer, and timing recovery
  3. Descrambler achieves lock → scr_status = OK
  4. Receiver signals loc_rcvr_status = OK to remote PHY via idle encoding
  5. Both sides ready → transition to SEND_N (normal data mode)

4. 100BASE-T1 PCS (Clause 96) — PAM3 over Single Twisted-Pair

4.1 Overview

100BASE-T1 operates at 100 Mb/s over a single balanced twisted-pair up to 15 m. It uses full-duplex communication with echo cancellation and 3-level PAM (PAM3) at 66.666 MBd.

  • Data rate: 100 Mb/s at MII
  • Symbol rate: 66.666 MBd (15 ns symbol period)
  • Modulation: PAM3 (ternary: –1, 0, +1)
  • Coding: 4B/3B conversion + side-stream scrambling + ternary pair mapping
  • Interface: Clause 22 MII (4-bit nibble-wide)

4.2 4B/3B Conversion

The PCS converts 4-bit MII nibbles at 25 MHz into 3-bit words at 33.333 MHz, maintaining the same bit throughput:

4 bits × 25 MHz = 100 Mb/s
3 bits × 33.333 MHz = 100 Mb/s

When the packet bit count is not a multiple of 3, stuff bits are appended:

  • 3n bits: No stuff bits
  • 3n+1 bits: 2 stuff bits appended
  • 3n+2 bits: 1 stuff bit appended

The stuff bits are discarded at the receiver during 3B/4B conversion. The minimum 12-byte IPG ensures FIFO does not overflow.

4.3 Side-Stream Scrambler

100BASE-T1 uses the same side-stream scrambler polynomials as 1000BASE-T (Clause 40.3.1.3.1):

RolePolynomial
MASTERgM(x) = 1 + x13 + x33
SLAVEgS(x) = 1 + x20 + x33

For 100BASE-T1, only 3 bits are used from the scrambler: Syn[2:0] and Scn[2:0]. Syn[3] is unused by definition.

4.12 Automatic Polarity Detection

During training, the SLAVE PHY can detect and correct polarity inversion:

  1. MASTER sends known training symbols
  2. SLAVE assumes one polarity and attempts descrambler lock
  3. If lock fails within timeout, SLAVE inverts received symbols (RAn, RBn) and tries again
  4. Once lock is achieved, SLAVE also inverts its transmitted symbols (TAn, TBn)
  5. MASTER always sees correct polarity because SLAVE has self-corrected

5. 1000BASE-T1 PCS (Clause 97) — RS-FEC & PAM3 @ 750 MBd

5.1 Overview

1000BASE-T1 achieves 1 Gb/s over a single twisted-pair copper cable for automotive/industrial applications. It employs:

  • Symbol rate: 750 MBd (4/3 ns symbol period ≈ 1.333 ns)
  • Modulation: PAM3 (ternary: –1, 0, +1)
  • Encoding: 80B/81B block coding + RS-FEC(450,406) + 15-bit scrambling
  • FEC: Reed-Solomon (450,406) over GF(29), correcting up to 22 symbol errors
  • Frame size: 4050 bits = 450 × 9-bit RS symbols = 2700 PAM3 symbols

5.2 80B/81B Block Encoding

Ten GMII transfers (80 bits of data/control) are encoded into an 81-bit block:

  • Bit 0: Data/Ctrl header (0 = data block, 1 = control block)

The pointer field indicates the position of the first control character:

5.3 RS-FEC Encoder (450,406)

The RS-FEC code is a shortened Reed-Solomon code over GF(29):

  • Code parameters: (450, 406) — 450 total symbols, 406 information symbols
  • Parity symbols: 44 (450 – 406)
  • Error correction: Up to 22 symbol errors (t = 22)
  • Field: GF(29) with primitive polynomial x9 + x4 + 1
  • Generator polynomial root: α = 0x002

Generator polynomial:

G(Z) = ∏(i=0 to 43) (Z – α^i) = A44·Z^44 + A43·Z^43 + ... + A1·Z + A0

Parity vector:

P(Z) = D(Z) · Z^44 mod G(Z)

Where D(Z) is the data vector of 406 symbols and P(Z) is the 44-symbol parity vector.

RTL Implementation: RS-FEC Encoder

// Galois Field multiplier for GF(2^9) with primitive poly x^9 + x^4 + 1
function [8:0] gf_mul(input [8:0] a, input [8:0] b);
  logic [17:0] p;
  p = a * b;  // polynomial multiplication
  // modular reduction by x^9 + x^4 + 1
  for (int i = 17; i >= 9; i--)
    if (p[i]) p = p ^ ({9'b0, 1'b1, 4'b0, 1'b1} << (i-9));
  return p[8:0];
endfunction

// LFSR-based RS encoder
always_ff @(posedge clk) begin
  if (rs_reset) begin
    shift_reg <= 44*{9'b0};
  end else if (rs_data_valid) begin
    feedback <= rs_data_in ^ shift_reg[43];
    for (int i = 43; i > 0; i--)
      shift_reg[i] <= gf_mul(feedback, A[i]) ^ shift_reg[i-1];
    shift_reg[0] <= gf_mul(feedback, A[0]);
  end
end

5.7 Training Mode

Training uses PAM2 (binary: –1, +1) instead of PAM3:

  • Training pattern: periodic embedded data in PAM2 stream
  • Alignment: 1-bit pattern every 180 PAM2 symbols, aligned with partial PHY frame boundary
  • InfoField: inserted in the 15th partial PHY frame
  • Partial PHY frame: 180 bits long; 15 partial frames = 1 PHY frame (2700 bits)
  • Training scrambler: 33-bit LFSR (same as 1000BASE-T/100BASE-T1)

Training bit generation:

Sn = Scrn[0] ⊕ InfoField[n mod 180]  if (n mod 2700) ≤ 96 and (n mod 180) == 0

   = Scrn[0]                           otherwise

Tn = +1 if Sn == 0
   = –1 if Sn == 1

EEE-capable PHYs use a quiet-refresh cycle:

A.1 1000BASE-X Synchronization State Machine — Full Detail

The PCS Synchronization process in 1000BASE-X manages code-group alignment and declares sync_status. The complete state machine from IEEE 802.3-2022 Clause 36 is reproduced below with implementation notes:

A.5 100BASE-T1 Differential Manchester Encoding (DME) for Training

100BASE-T1 uses Differential Manchester Encoding during Auto-Negotiation (Clause 98) to exchange capabilities before transitioning to PAM3 data mode:

  • DME clock: Transitions at the center of every bit cell
  • DME data: A transition at the cell boundary represents logic 1; no transition represents logic 0
  • Bit rate: 66.666 Mb/s (same symbol rate as PAM3)
  • Purpose: Half-duplex negotiation; does not require echo cancellation

DME encoding rules:

Previous StateData BitOutput Transition
High0Fall at center only
High1Fall at start, rise at center
Low0Rise at center only
Low1Rise at start, fall at center

A.6 1000BASE-T1 RS-FEC Syndrome Calculation

The RS-FEC decoder computes syndromes S0 through S43 by evaluating the received polynomial R(Z) at αi:

Si = R(α^i) for i = 0, 1, ..., 43

If all Si = 0, the received frame is error-free. Otherwise, the Berlekamp-Massey algorithm computes the error locator polynomial, and the Chien search finds error locations. Forney's algorithm computes error magnitudes.

RTL: Syndrome Calculator (Simplified)

module rs_syndrome (
  input  wire        clk,
  input  wire        rst_n,
  input  wire  [8:0] rx_symbol,
  input  wire        symbol_valid,
  input  wire        frame_start,
  output logic [8:0] syndrome [0:43]
);
  // GF(2^9) multiplier and accumulator
  logic [8:0] acc [0:43];
  logic [8:0] alpha_power [0:43];  // Pre-computed α^i
  
  always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      for (int i = 0; i < 44; i++) begin
        acc[i] <= 9'd0;
        syndrome[i] <= 9'd0;
      end
    end else begin
      if (frame_start) begin
        for (int i = 0; i < 44; i++) acc[i] <= 9'd0;
      end
      if (symbol_valid) begin
        for (int i = 0; i < 44; i++) begin
          acc[i] <= gf_mul(acc[i], alpha_power[i]) ^ rx_symbol;
        end
      end
      if (frame_end) begin
        for (int i = 0; i < 44; i++) syndrome[i] <= acc[i];
      end
    end
  end
endmodule

A.7 Link Synchronization (1000BASE-T1 without Auto-Negotiation)

When Auto-Negotiation is disabled, 1000BASE-T1 uses Link Synchronization:

  1. MASTER sends synchronization sequence for 1 µs
  2. If no SLAVE response, MASTER repeats every 5 µs
  3. SLAVE detects sequence, responds with 1 µs synchronization sequence
  4. If no other detection for 4 µs after SLAVE response, Link Synchronization is complete
  5. Link Monitor timers start; PHY Control begins Training
// MASTER Link Synchronization
MASTER_SEND:
  tx_mode <= SEND_T;  // PAM2 training
  send_sync_sequence(1us);
  wait 5us;
  if (!slave_detected) goto MASTER_SEND;
  else goto LINK_SYNC_DONE;

// SLAVE Link Synchronization  
SLAVE_LISTEN:
  if (sync_sequence_detected) goto SLAVE_RESPOND;

SLAVE_RESPOND:
  send_sync_sequence(1us);
  wait 4us;
  if (no_other_detection) goto LINK_SYNC_DONE;

A.8 100BASE-X PMA Carrier Detect State Machine

// States: INITIALIZE, RECEIVE_NEXT_BIT, CARRIER_DETECT, CARRIER_OFF,
//         WAIT_FOR_NEXT, GET_NEXT_QUINT, CONFIRM_K, BAD_CARRIER

INITIALIZE:
  r_bits[9:0] <= 10'b11111_11111;
  carrier_status <= OFF;
  rxerror_status <= NO_ERROR;
  goto RECEIVE_NEXT_BIT;

RECEIVE_NEXT_BIT:
  SHIFTLEFT(r_bits);
  r_bits[0] <= rx_code_bit;
  if (r_bits[0] == 0 && r_bits[9:2] != 8'b11111111)
    goto CARRIER_DETECT;  // Two non-contiguous zeros in 10 bits

CARRIER_DETECT:
  carrier_status <= ON;
  rxerror_status <= NO_ERROR;
  // Look for SSD pattern 11000_10001 in next 10 bits
  if (r_bits[9:0] == 10'b11000_10001)
    goto GET_NEXT_QUINT;
  else if (r_bits[9:0] == 10'b11111_11000)
    goto CONFIRM_K;
  else
    goto RECEIVE_NEXT_BIT;

CONFIRM_K:
  rxerror_status <= NO_ERROR;
  goto GET_NEXT_QUINT;

GET_NEXT_QUINT:
  // Wait for 5 more bits
  5xPMD_UNITDATA.indicates;
  if (r_bits[9:0] == 10'b11000_10001)
    goto WAIT_FOR_NEXT;
  else
    goto BAD_CARRIER;

BAD_CARRIER:
  rxerror_status <= ERROR;
  if (carrier_status == ON) goto RECEIVE_NEXT_BIT;
  else goto CARRIER_OFF;

CARRIER_OFF:
  carrier_status <= OFF;
  if (r_bits[9:0] == 10'b11111_11111) goto INITIALIZE;
  else goto RECEIVE_NEXT_BIT;

WAIT_FOR_NEXT:
  if (link_status != OK) goto INITIALIZE;
  goto RECEIVE_NEXT_BIT;

A.9 1000BASE-T PHY Control State Machine

// States: SILENT, TRAINING, ANALYZE, LINK_OK, SEND_IDLE_OR_DATA,
//         SEND_I, SEND_N, SEND_Z

SILENT:
  tx_mode <= SEND_Z;
  if (link_control == ENABLE) goto TRAINING;

TRAINING:
  tx_mode <= SEND_I;
  start minwait_timer;
  if (minwait_timer_done && scr_status == OK)
    goto ANALYZE;

ANALYZE:
  tx_mode <= SEND_I;
  if (loc_rcvr_status == OK && rem_rcvr_status == OK)
    goto LINK_OK;
  else if (maxwait_timer_done)
    goto SILENT;

LINK_OK:
  tx_mode <= SEND_N;
  if (loc_rcvr_status == NOT_OK || rem_rcvr_status == NOT_OK)
    goto TRAINING;

SEND_IDLE_OR_DATA:
  if (tx_enable) goto SEND_N;
  else goto SEND_I;

A.10 100BASE-T1 4B/3B and 3B/4B Conversion Detail

The 4B/3B conversion packs 4-bit nibbles into 3-bit words:

// 4B/3B conversion example (3n bits, no stuff bits)
// d0<3:0> d1<3:0> d2<3:0> → d0<2:0> d1<1:0>,d0<3> d2<0>,d1<3:2> d2<3:1>
//
// 4B/3B conversion (3n+1 bits, 2 stuff bits)
// d0<3:0> d1<3:0> → d0<2:0> d1<1:0>,d0<3> ,d1<3>  (2 stuff bits appended)
//
// 4B/3B conversion (3n+2 bits, 1 stuff bit)
// d0<3:0> d1<3:0> d2<3:0> → d0<2:0> d1<1:0>,d0<3> d2<0>,d1<3:2> d2<3:1>
// ... d3<3:0> → d3<2:0> ,d3<3>  (1 stuff bit appended)

The 3B/4B conversion at the receiver reverses this, discarding stuff bits:

// 3B/4B conversion
// rx_data<2:0> at 33.333 MHz → RXD<3:0> at 25 MHz
// Accumulate 3 bits → form nibbles → output on MII
// If residual bits remain at end of packet (stuff bits), discard them
