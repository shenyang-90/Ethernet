// Working in Ideal (per CTS) mode so this is required for Ideal clocks
set rule handling sdc_def_checks/sdc_clock_checks/CCD_CLK_CTR8 -severity Note    ; //Default severity (Warning) -- set_clock_transition is set on a clock in post-layout

// Set input transition is not required - clocks are ideal (pre-cts),transition time is set using set_clock_transition
set rule handling sdc_def_checks/sdc_clock_checks/CCD_CLK_CTR6 -severity Note    ; //Default severity (Warning) -- Undefined clock transition on real clock: Source input port without set_input_transition

// Don't want to prevent optimisation on clock nets (clock gate insertion ,etc)
set rule handling sdc_def_checks/sdc_design_checks/CCD_MISC_HFN1 -severity Note   ;  //Clock is not set as dont_touch_network (pre-layout)

//Not valid at the IP block level
set rule handling sdc_def_checks/sdc_design_checks/CCD_MISC_HFN1b -severity Note   ; //Set/reset is not set as dont_touch_network

// Not applicable at the pre-layout stage
set rule handling sdc_def_checks/sdc_design_checks/CCD_MISC_HFN10 -severity Note   ; //Invalid constraint in current context (post-layout)

// Not valid for pre-layout checks
set rule handling sdc_def_checks/sdc_design_checks/CCD_DGN_PRT6  -severity Note   ; //Unbuffered net connection to output port (post-layout)

// No issue for clock name to be the same as the port 
set rule handling sdc_def_checks/sdc_design_checks/CCD_MISC_NAM1  -severity Note   ; //Clock has same name as port or pin

// Not required at block level
set rule handling sdc_def_checks/sdc_design_checks/CCD_MISC_POW1  -severity Note   ; //Undefined maximum dynamic power constraint
set rule handling sdc_def_checks/sdc_design_checks/CCD_MISC_POW2  -severity Note   ; //Undefined maximum leakage power constraint
